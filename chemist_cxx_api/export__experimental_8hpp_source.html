58inline void export_experimental(python_module_reference m) {
59 auto sub = m.def_submodule(
60 "experimental",
61 "Classes which are still being designed. Their APIs may change.");
62 export_point_class(sub);
63 export_point_view(sub);
64 export_point_set_class(sub);
65 export_point_set_view(sub);
66 export_primitive_class(sub);
67 export_primitive_view(sub);
68 export_contracted_gaussian_class(sub);
69 export_contracted_gaussian_view(sub);
70 // The abstract bases must be registered before the classes deriving from
71 // them, since pybind11 needs the base to exist to record the inheritance.
72 export_ao(sub);
73 export_ao_shell(sub);
74 export_cartesian_ao_class(sub);
75 export_cartesian_ao_view(sub);
76 export_spherical_ao_class(sub);
77 export_spherical_ao_view(sub);
78 export_cca_shell_class(sub);

108template<typename FloatViewType>
109double to_py_float(const FloatViewType& value) {
110 try {
111 return value.template value<double>();
112 } catch(const std::runtime_error&) {
113 throw py::type_error(
114 "This coordinate holds a " + value.type_info().name() +
115 ", which Python can not represent. Only double is supported at "
116 "the Python boundary.");
117 }
118}

128inline double to_py_float(const wtf::fp::Float& value) {
129 return to_py_float(value.as_view());
130}

138template<typename PyClass>
139void add_point_readers(PyClass& c) {
140 using class_type = typename PyClass::type;
141 c.def("get_coord",
142 [](const class_type& p, std::size_t q) {
143 return to_py_float(p.get_coord(q));
144 })
145 .def("get_x", [](const class_type& p) { return to_py_float(p.get_x()); })
146 .def("get_y", [](const class_type& p) { return to_py_float(p.get_y()); })
147 .def("get_z", [](const class_type& p) { return to_py_float(p.get_z()); })
148 .def("magnitude",
149 [](const class_type& p) { return to_py_float(p.magnitude()); })
150 .def("__repr__", [](const class_type& p) {
151 std::stringstream ss;
152 ss << p;
153 return ss.str();
154 });
155}

268template<typename PyClass>
269void add_set_readers(PyClass& c) {
270 using class_type = typename PyClass::type;
271 c.def("__len__", [](const class_type& s) { return s.size(); })
272 // Note that this takes a mutable reference: iterating a mutable set
273 // must yield mutable views of its points, exactly as the C++
274 // range-for does. When class_type is itself read-only, its iterators
275 // hand out read-only views anyway.
276 .def(
277 "__iter__",
278 [](class_type& s) { return py::make_iterator(s.begin(), s.end()); },
279 py::keep_alive<0, 1>())
280 .def("empty", [](const class_type& s) { return s.empty(); })
281 .def("get_x_coordinates",
282 [](const class_type& s) {
283 std::vector<double> rv;
284 const auto n = s.size();
285 for(std::size_t i = 0; i < n; ++i)
286 rv.push_back(to_py_float(s[i].get_x()));
287 return rv;
288 })
289 .def("get_y_coordinates",
290 [](const class_type& s) {
291 std::vector<double> rv;
292 const auto n = s.size();
293 for(std::size_t i = 0; i < n; ++i)
294 rv.push_back(to_py_float(s[i].get_y()));
295 return rv;
296 })
297 .def("get_z_coordinates", [](const class_type& s) {
298 std::vector<double> rv;
299 const auto n = s.size();
300 for(std::size_t i = 0; i < n; ++i)
301 rv.push_back(to_py_float(s[i].get_z()));
302 return rv;
303 });
304}

322template<typename ContainerType>
323std::vector<double> to_py_floats(const ContainerType& values) {
324 std::vector<double> rv;
325 rv.reserve(values.size());
326 for(const auto& value : values) rv.push_back(to_py_float(value));
327 return rv;
328}

343template<typename PyClass>
344void add_ao_readers(PyClass& c) {
345 using class_type = typename PyClass::type;
346 c.def("get_l", [](const class_type& ao) { return ao.get_l(); })
347 .def(
348 "get_center", [](const class_type& ao) { return ao.get_center(); },
349 py::keep_alive<0, 1>())
350 // Takes a mutable reference so that, when class_type models mutable
351 // state, the returned view is mutable too, exactly as in C++.
352 .def(
353 "get_contracted_gaussian",
354 [](class_type& ao) { return ao.get_contracted_gaussian(); },
355 py::keep_alive<0, 1>())
356 .def("normalization_constant",
357 [](const class_type& ao) {
358 return to_py_float(ao.normalization_constant());
359 })
360 .def("evaluate",
361 [](const class_type& ao, const const_point_view& r) {
362 return to_py_float(ao.evaluate(r));
363 })
364 .def("evaluate",
365 [](const class_type& ao, const const_point_set_view& points) {
366 return to_py_floats(ao.evaluate(points));
367 })
368 .def("normalized_evaluate",
369 [](const class_type& ao, const const_point_view& r) {
370 return to_py_float(ao.normalized_evaluate(r));
371 })
372 .def("normalized_evaluate",
373 [](const class_type& ao, const const_point_set_view& points) {
374 return to_py_floats(ao.normalized_evaluate(points));
375 });

459template<typename PyClass>
460void add_cca_shell_readers(PyClass& c) {
461 using class_type = typename PyClass::type;
462 using shell_traits = ChemistClassTraits<typename class_type::value_type>;
463 add_ao_shell_readers(c);
464
465 auto at = [](const class_type& s, std::size_t i) { return s.at(i); };
466 c.def("at", at, py::keep_alive<0, 1>())
467 .def("__getitem__", at, py::keep_alive<0, 1>())
468 .def(
469 "get_cartesian_shell",
470 [](const class_type& s) { return s.get_cartesian_shell(); },
471 py::keep_alive<0, 1>());
472
473 if constexpr(shell_traits::is_pure) {
474 c.def("magnetic_index", [](const class_type& s, std::size_t i) {
475 return s.magnetic_index(i);
476 });
477 } else {
478 c.def("cartesian_powers", [](const class_type& s, std::size_t i) {
479 const auto [x, y, z] = s.cartesian_powers(i);
480 return py::make_tuple(x, y, z);
481 });
482 }
483}

501template<typename OtherType, typename PyClass>

503 using class_type = typename PyClass::type;

505 const OtherType& rhs) { return lhs == rhs; })
506 .def("__ne__", [](const class_type& lhs, const OtherType& rhs) {
