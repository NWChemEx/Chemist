83template<typename FloatViewType>
84double to_py_float(const FloatViewType& value) {
85 try {
86 return value.template value<double>();
87 } catch(const std::runtime_error&) {
88 throw py::type_error(
89 "This coordinate holds a " + value.type_info().name() +
90 ", which Python can not represent. Only double is supported at "
91 "the Python boundary.");
92 }
93}

113template<typename PyClass>
114void add_point_readers(PyClass& c) {
115 using class_type = typename PyClass::type;
116 c.def("get_coord",
117 [](const class_type& p, std::size_t q) {
118 return to_py_float(p.get_coord(q));
119 })
120 .def("get_x", [](const class_type& p) { return to_py_float(p.get_x()); })
121 .def("get_y", [](const class_type& p) { return to_py_float(p.get_y()); })
122 .def("get_z", [](const class_type& p) { return to_py_float(p.get_z()); })
123 .def("magnitude",
124 [](const class_type& p) { return to_py_float(p.magnitude()); })
125 .def("__repr__", [](const class_type& p) {
126 std::stringstream ss;
127 ss << p;
128 return ss.str();
129 });
130}

153template<typename PyClass>
154void add_gaussian_readers(PyClass& c) {
155 using class_type = typename PyClass::type;
156 c.def("get_l", [](const class_type& g) { return g.get_l(); })
157 .def("get_center", [](const class_type& g) { return g.get_center(); })
158 .def("evaluate",
159 [](const class_type& g, const Point& r) {
160 return to_py_float(g.evaluate(r));
161 })
162 .def("normalized_evaluate",
163 [](const class_type& g, const Point& r) {
164 return to_py_float(g.normalized_evaluate(r));
165 })
166 .def("normalization_constant", [](const class_type& g) {
167 return to_py_float(g.normalization_constant());
168 });
169}

216template<typename PyClass>
217void add_contracted_gaussian_readers(PyClass& c) {
218 using class_type = typename PyClass::type;

243template<typename PyClass>
244void add_set_readers(PyClass& c) {
245 using class_type = typename PyClass::type;
246 c.def("__len__", [](const class_type& s) { return s.size(); })
247 // Note that this takes a mutable reference: iterating a mutable set
248 // must yield mutable views of its points, exactly as the C++
249 // range-for does. When class_type is itself read-only, its iterators
250 // hand out read-only views anyway.
251 .def(
252 "__iter__",
253 [](class_type& s) { return py::make_iterator(s.begin(), s.end()); },
254 py::keep_alive<0, 1>())
255 .def("empty", [](const class_type& s) { return s.empty(); })
256 .def("get_x_coordinates",
257 [](const class_type& s) {
258 std::vector<double> rv;
259 const auto n = s.size();
260 for(std::size_t i = 0; i < n; ++i)
261 rv.push_back(to_py_float(s[i].get_x()));
262 return rv;
263 })
264 .def("get_y_coordinates",
265 [](const class_type& s) {
266 std::vector<double> rv;
267 const auto n = s.size();
268 for(std::size_t i = 0; i < n; ++i)
269 rv.push_back(to_py_float(s[i].get_y()));
270 return rv;
271 })
272 .def("get_z_coordinates", [](const class_type& s) {
273 std::vector<double> rv;
274 const auto n = s.size();
275 for(std::size_t i = 0; i < n; ++i)
276 rv.push_back(to_py_float(s[i].get_z()));
277 return rv;
278 });
279}
