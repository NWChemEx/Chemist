121template<typename FloatViewType>
122double to_py_float(const FloatViewType& value) {
123 try {
124 return value.template value<double>();
125 } catch(const std::runtime_error&) {
126 throw py::type_error(
127 "This coordinate holds a " + value.type_info().name() +
128 ", which Python can not represent. Only double is supported at "
129 "the Python boundary.");
130 }
131}

151template<typename PyClass>
152void add_point_readers(PyClass& c) {
153 using class_type = typename PyClass::type;
154 c.def("get_coord",
155 [](const class_type& p, std::size_t q) {
156 return to_py_float(p.get_coord(q));
157 })
158 .def("get_x", [](const class_type& p) { return to_py_float(p.get_x()); })
159 .def("get_y", [](const class_type& p) { return to_py_float(p.get_y()); })
160 .def("get_z", [](const class_type& p) { return to_py_float(p.get_z()); })
161 .def("magnitude",
162 [](const class_type& p) { return to_py_float(p.magnitude()); })
163 .def("__repr__", [](const class_type& p) {
164 std::stringstream ss;
165 ss << p;
166 return ss.str();
167 });
168}

281template<typename PyClass>
282void add_set_readers(PyClass& c) {
283 using class_type = typename PyClass::type;
284 c.def("__len__", [](const class_type& s) { return s.size(); })
285 // Note that this takes a mutable reference: iterating a mutable set
286 // must yield mutable views of its points, exactly as the C++
287 // range-for does. When class_type is itself read-only, its iterators
288 // hand out read-only views anyway.
289 .def(
290 "__iter__",
291 [](class_type& s) { return py::make_iterator(s.begin(), s.end()); },
292 py::keep_alive<0, 1>())
293 .def("empty", [](const class_type& s) { return s.empty(); })
294 .def("get_x_coordinates",
295 [](const class_type& s) {
296 std::vector<double> rv;
297 const auto n = s.size();
298 for(std::size_t i = 0; i < n; ++i)
299 rv.push_back(to_py_float(s[i].get_x()));
300 return rv;
301 })
302 .def("get_y_coordinates",
303 [](const class_type& s) {
304 std::vector<double> rv;
305 const auto n = s.size();
306 for(std::size_t i = 0; i < n; ++i)
307 rv.push_back(to_py_float(s[i].get_y()));
308 return rv;
309 })
310 .def("get_z_coordinates", [](const class_type& s) {
311 std::vector<double> rv;
312 const auto n = s.size();
313 for(std::size_t i = 0; i < n; ++i)
314 rv.push_back(to_py_float(s[i].get_z()));
315 return rv;
316 });
317}

335template<typename ContainerType>
336std::vector<double> to_py_floats(const ContainerType& values) {
337 std::vector<double> rv;
338 rv.reserve(values.size());
339 for(const auto& value : values) rv.push_back(to_py_float(value));
340 return rv;

478template<typename PyClass>
479void add_cca_shell_readers(PyClass& c) {
480 using class_type = typename PyClass::type;
481 using shell_traits = ChemistClassTraits<typename class_type::value_type>;
482 add_ao_shell_readers(c);
483
484 auto at = [](const class_type& s, std::size_t i) { return s.at(i); };
485 c.def("at", at, py::keep_alive<0, 1>())
486 .def("__getitem__", at, py::keep_alive<0, 1>())
487 .def(
488 "get_cartesian_shell",
489 [](const class_type& s) { return s.get_cartesian_shell(); },
490 py::keep_alive<0, 1>());
491
492 if constexpr(shell_traits::is_pure) {
493 c.def("magnetic_index", [](const class_type& s, std::size_t i) {
494 return s.magnetic_index(i);
495 });
496 } else {
497 c.def("cartesian_powers", [](const class_type& s, std::size_t i) {
498 const auto [x, y, z] = s.cartesian_powers(i);
499 return py::make_tuple(x, y, z);
500 });
501 }
