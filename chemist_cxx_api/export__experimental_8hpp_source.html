73template<typename FloatViewType>
74double to_py_float(const FloatViewType& value) {
75 try {
76 return value.template value<double>();
77 } catch(const std::runtime_error&) {
78 throw py::type_error(
79 "This coordinate holds a " + value.type_info().name() +
80 ", which Python can not represent. Only double is supported at "
81 "the Python boundary.");
82 }
83}
84
93inline double to_py_float(const wtf::fp::Float& value) {
94 return to_py_float(value.as_view());
95}

103template<typename PyClass>
104void add_point_readers(PyClass& c) {
105 using class_type = typename PyClass::type;
106 c.def("get_coord",
107 [](const class_type& p, std::size_t q) {
108 return to_py_float(p.get_coord(q));
109 })
110 .def("get_x", [](const class_type& p) { return to_py_float(p.get_x()); })
111 .def("get_y", [](const class_type& p) { return to_py_float(p.get_y()); })
112 .def("get_z", [](const class_type& p) { return to_py_float(p.get_z()); })
113 .def("magnitude",
114 [](const class_type& p) { return to_py_float(p.magnitude()); })
115 .def("__repr__", [](const class_type& p) {
116 std::stringstream ss;
117 ss << p;
118 return ss.str();
119 });
120}

123template<typename PyClass>
124void add_point_writers(PyClass& c) {
125 using class_type = typename PyClass::type;
126 c.def("set_coord",
127 [](class_type& p, std::size_t q, double v) { p.set_coord(q, v); })
128 .def("set_x", [](class_type& p, double v) { p.set_x(v); })
129 .def("set_y", [](class_type& p, double v) { p.set_y(v); })
130 .def("set_z", [](class_type& p, double v) { p.set_z(v); });
131}

141template<typename PyClass>
142void add_set_readers(PyClass& c) {
143 using class_type = typename PyClass::type;
144 c.def("__len__", [](const class_type& s) { return s.size(); })
145 // Note that this takes a mutable reference: iterating a mutable set
146 // must yield mutable views of its points, exactly as the C++ range-for
147 // does. When class_type is itself read-only, its iterators hand out
148 // read-only views anyway.
149 .def(
150 "__iter__",
151 [](class_type& s) { return py::make_iterator(s.begin(), s.end()); },
152 py::keep_alive<0, 1>())
153 .def("empty", [](const class_type& s) { return s.empty(); })
154 .def("get_x_coordinates",
155 [](const class_type& s) {
156 std::vector<double> rv;
157 const auto n = s.size();
158 for(std::size_t i = 0; i < n; ++i)
159 rv.push_back(to_py_float(s[i].get_x()));
160 return rv;
161 })
162 .def("get_y_coordinates",
163 [](const class_type& s) {
164 std::vector<double> rv;
165 const auto n = s.size();
166 for(std::size_t i = 0; i < n; ++i)
167 rv.push_back(to_py_float(s[i].get_y()));
168 return rv;
169 })
170 .def("get_z_coordinates", [](const class_type& s) {
171 std::vector<double> rv;
172 const auto n = s.size();
173 for(std::size_t i = 0; i < n; ++i)
174 rv.push_back(to_py_float(s[i].get_z()));
175 return rv;
176 });
177}
