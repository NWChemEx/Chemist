120template<typename T>
121bool is_exactly(const py::object& o) {
122 return py::type::of(o).is(py::type::of<T>());
123}

131template<typename T>
132bool particle_matches(const py::object& o, bool exact) {
133 return exact ? is_exactly<T>(o) : py::isinstance<T>(o);
134}

137template<typename Table, typename Make, std::size_t... I>
138py::object select_pair_(const py::object& lhs, const py::object& rhs,
139 bool exact, Make&& make, std::index_sequence<I...>) {
140 py::object result = py::none();
141 auto try_one = [&](auto tag) {
142 using T = typename decltype(tag)::lhs_type;
143 using U = typename decltype(tag)::rhs_type;
144 if(!result.is_none()) return;
145 if(particle_matches<T>(lhs, exact) && particle_matches<U>(rhs, exact))
146 result = make(tag);
147 };
148 (try_one(std::tuple_element_t<I, Table>{}), ...);
149 return result;
150}

179py::object select_type_(const py::object& particle, bool exact, Make&& make,
180 std::index_sequence<I...>) {

209template<template<typename...> class Op, typename Customize, typename T>
210py::object export_instantiation_(python_module_reference m, const char* op_name,
211 Customize& customize, type_tag<T> tag) {
212 using op_type = Op<T>;
213
214 std::string name("_");
215 name += op_name;
216 name += particle_traits<T>::suffix;
217
218 auto get_particle = [](const op_type& o) { return o.get_particle(); };
219 auto set_particle = [](op_type& o, T& p) { o.set_particle(p); };
220
221 python_class_type<op_type, OperatorBase, py::smart_holder> cls(
222 m, name.c_str());
223 cls.def(py::init<>())
224 .def(py::self == py::self)
225 .def(py::self != py::self)
226 .def_property("particle", get_particle, set_particle);
227
228 customize(cls, tag);
229 return cls;
230}

403template<template<typename...> class Op, typename LeadingArgs, typename T,
404 typename U, std::size_t... L>
405py::object construct_(const py::args& args, const py::object& lhs,
406 const py::object& rhs, std::index_sequence<L...>) {
407 return py::cast(
408 Op<T, U>(args[L].template cast<std::tuple_element_t<L, LeadingArgs>>()...,
409 lhs.template cast<T>(), rhs.template cast<U>()));
410}

436template<template<typename...> class Op, typename Table, typename DefaultPair,
437 typename LeadingArgs = std::tuple<>>
438py::cpp_function make_two_particle_dispatch(const char* name,
439 const char* arg_error) {
440 constexpr auto n_leading = std::tuple_size_v<LeadingArgs>;
441
442 // N.B. The leading parameter is the class CPython passes to __new__. It is
443 // always the dispatching class itself, so it is ignored.
444 auto impl = [name, arg_error](py::object, py::args args,
445 py::kwargs kwargs) -> py::object {
446 assert_no_kwargs(name, kwargs);
447
448 // No arguments gives the same object as the C++ default ctor.
449 if(args.size() == 0)
450 return py::cast(Op<typename DefaultPair::lhs_type,
451 typename DefaultPair::rhs_type>{});
452
453 if(args.size() != n_leading + 2)
454 throw py::type_error(std::string(name) + " " + arg_error + ".");
455
456 auto lhs = py::reinterpret_borrow<py::object>(args[n_leading]);
457 auto rhs = py::reinterpret_borrow<py::object>(args[n_leading + 1]);
458
459 auto make = [&](auto tag) {
460 using T = typename decltype(tag)::lhs_type;
461 using U = typename decltype(tag)::rhs_type;
462 return construct_<Op, LeadingArgs, T, U>(
463 args, lhs, rhs, std::make_index_sequence<n_leading>{});
464 };
465
466 auto result = select_pair<Table>(lhs, rhs, make);
467 if(!result.is_none()) return result;
468
469 throw py::type_error(
470 std::string(name) + " can not describe the interaction of a " +
471 python_type_name(lhs) + " with a " + python_type_name(rhs) +
472 ". Supported combinations are: " + supported_combinations<Table>() +
473 ".");
474 };
475
476 return py::cpp_function(impl);
477}
