218 template<typename OtherDerived, typename OtherPoint>
219 coord_type inner_product(
220 const PointCommon<OtherDerived, OtherPoint>& rhs) const {
221 auto xx = detail_::multiply(get_x(), rhs.get_x());
222 auto yy = detail_::multiply(get_y(), rhs.get_y());
223 auto zz = detail_::multiply(get_z(), rhs.get_z());
224 auto xy = detail_::add(as_const_view_(xx), as_const_view_(yy));
225 return detail_::add(as_const_view_(xy), as_const_view_(zz));
226 }

240 coord_type magnitude() const {
241 auto ip = inner_product(*this);

243 }
