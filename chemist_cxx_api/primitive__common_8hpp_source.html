259 template<typename OtherDerived, typename OtherPoint>
260 numerical_value evaluate(
261 const PointCommon<OtherDerived, OtherPoint>& r) const {
262 auto dr = r - get_center();
263 const coord_type r2 = dr.inner_product(dr);
264 auto exponent_term = detail_::multiply(get_exponent(), as_view_(r2));
265 auto neg_term = detail_::negate(as_view_(exponent_term));
266 auto gaussian = detail_::exp(as_view_(neg_term));
267 return detail_::multiply(get_coefficient(), as_view_(gaussian));
268 }

284 template<typename OtherDerived, typename OtherPointSet>
285 numerical_vector evaluate(
286 const PointSetCommon<OtherDerived, OtherPointSet>& points) const {
287 numerical_vector rv;
288 const auto n = points.size();
289 rv.reserve(n);
290 for(size_type i = 0; i < n; ++i) rv.push_back(evaluate(points[i]));
291 return rv;
292 }

339 template<typename OtherDerived, typename OtherPointSet>
340 numerical_vector normalized_evaluate(
341 const PointSetCommon<OtherDerived, OtherPointSet>& points) const {
342 auto n_chi = normalization_constant();
343 numerical_vector rv;
344 const auto n = points.size();
345 rv.reserve(n);
346 for(size_type i = 0; i < n; ++i) {
347 auto value = evaluate(points[i]);
348 rv.push_back(detail_::multiply(as_view_(n_chi), as_view_(value)));
