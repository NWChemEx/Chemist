407 template<typename ValueFxn>
408 numerical_value transform_(ValueFxn&& value_of,
409 bool divide_out_n_ao) const {
410 const auto& shell = downcast_().shell_();
411 const auto l = shell.get_l();
412 const auto m = get_m();
413
414 // The sum is over every (i, j, k) with i + j + k = l, and does not
415 // depend on the order they are visited in, so they are enumerated here
416 // rather than asked of the shell, whose ordering *this does not know.
417 // Each Cartesian AO is built directly on the shell's contracted
418 // Gaussian for the same reason.
419 std::optional<numerical_value> sum;
420 for(size_type i = 0; i <= l; ++i) {
421 for(size_type j = 0; i + j <= l; ++j) {
422 const size_type k = l - i - j;
423 auto c =
424 detail_::spherical_transform_coefficient(l, m, i, j, k);
425 if(c == 0.0) continue;
426 if(divide_out_n_ao)
427 c /= detail_::cartesian_ao_normalization(i, j, k);
428
429 const const_cartesian_ao_reference ao(
430 shell.get_contracted_gaussian(), i, j, k);
431 const auto ao_value = value_of(ao);
432 auto term = detail_::scale(ao_value.as_view(), c);
433 if(sum.has_value())
434 sum = detail_::add(sum->as_view(), term.as_view());
435 else
436 sum = std::move(term);
437 }
438 }
439 if(!sum.has_value())
440 throw std::logic_error(
441 "chemist::experimental::SphericalAO: no Cartesian AO contributes "
442 "to this component. This should not be possible.");
443 return std::move(*sum);
444 }
