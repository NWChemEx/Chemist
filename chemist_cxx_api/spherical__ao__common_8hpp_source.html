407 template<typename ValueFxn>
408 numerical_value transform_(ValueFxn&& value_of) const {
409 const auto& shell = downcast_().shell_();
410 const auto l = shell.get_l();
411 const auto m = get_m();
412
413 // The sum is over every (i, j, k) with i + j + k = l, and does not
414 // depend on the order they are visited in, so they are enumerated here
415 // rather than asked of the shell, whose ordering *this does not know.
416 // Each Cartesian AO is built directly on the shell's contracted
417 // Gaussian for the same reason.
418 std::optional<numerical_value> sum;
419 for(size_type i = 0; i <= l; ++i) {
420 for(size_type j = 0; i + j <= l; ++j) {
421 const size_type k = l - i - j;
422 const auto c =
423 detail_::spherical_transform_coefficient(l, m, i, j, k);
424 if(c == 0.0) continue;
425
426 const const_cartesian_ao_reference ao(
427 shell.get_contracted_gaussian(), i, j, k);
428 const auto ao_value = value_of(ao);
429 auto term = detail_::scale(ao_value.as_view(), c);
430 if(sum.has_value())
431 sum = detail_::add(sum->as_view(), term.as_view());
432 else
433 sum = std::move(term);
434 }
435 }
436 if(!sum.has_value())
437 throw std::logic_error(
438 "chemist::experimental::SphericalAO: no Cartesian AO contributes "
439 "to this component. This should not be possible.");
440 return std::move(*sum);
441 }
