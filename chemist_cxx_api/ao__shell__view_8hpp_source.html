34template<typename AOType>
35class AOShellView : public AOShellCommon<AOShellBaseView, AOType> {
36private:
38 using base_type = AOShellCommon<AOShellBaseView, AOType>;
39
40public:
43 using pointer = std::unique_ptr<AOShellView>;
44
47 using shell_pointer = typename AOShell<AOType>::pointer;
48
50 ~AOShellView() noexcept override = default;
51
52 // -------------------------------------------------------------------------
53 // -- Utility
54 // -------------------------------------------------------------------------
55
66 pointer clone() const {
67 // Every class deriving from *this clones into its own type, which is
68 // an AOShellView<AOType>, so the downcast can not fail.
69 return pointer(static_cast<AOShellView*>(this->clone_().release()));
70 }
71
82 shell_pointer as_shell() const {
83 // A view materializes into the shell it is a view of, which has the
84 // same purity, so the downcast can not fail.
85 return shell_pointer(
86 static_cast<AOShell<AOType>*>(this->as_shell_().release()));
87 }
88
89protected:
92 AOShellView() noexcept = default;
93 AOShellView(const AOShellView&) noexcept = default;
94 AOShellView(AOShellView&&) noexcept = default;
95 AOShellView& operator=(const AOShellView&) noexcept = default;
96 AOShellView& operator=(AOShellView&&) noexcept = default;
98};
