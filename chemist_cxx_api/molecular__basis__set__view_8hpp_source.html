155 MolecularBasisSetView(buffer_reference coefficients,
156 buffer_reference exponents, l_span ls,
157 offset_span primitive_offsets,
158 offset_span shell_offsets,
159 center_set_reference centers, name_span names,
160 atomic_number_span atomic_numbers,
161 purity_span purities, ordering_span orderings) :
162 m_coefficients_(std::move(coefficients)),
163 m_exponents_(std::move(exponents)),
164 m_ls_(ls),
165 m_primitive_offsets_(primitive_offsets),
166 m_shell_offsets_(shell_offsets),
167 m_centers_(std::move(centers)),
168 m_names_(names),
169 m_atomic_numbers_(atomic_numbers),
170 m_purities_(purities),
171 m_orderings_(orderings) {
172 const auto n_atoms = m_names_.size();
173 if(m_shell_offsets_.size() != n_atoms + 1 ||
174 m_centers_.size() != n_atoms ||
175 m_atomic_numbers_.size() != n_atoms ||
176 m_purities_.size() != n_atoms || m_orderings_.size() != n_atoms)
177 throw_bad_state_("there must be one center, atomic number, "
178 "purity, and AO ordering per name, and one more "
179 "atom offset than there are names");
180 if(m_shell_offsets_.back() - m_shell_offsets_.front() != m_ls_.size() ||
181 m_primitive_offsets_.size() != m_ls_.size() + 1)
182 throw_bad_state_("the atom offsets must describe exactly the "
183 "shells, and there must be one more shell offset "
184 "than there are shells");
185 const auto n =
186 m_primitive_offsets_.back() - m_primitive_offsets_.front();
187 if(m_coefficients_.size() != n || m_exponents_.size() != n)
188 throw_bad_state_("the parameter arrays must hold exactly the "
189 "primitives the shell offsets describe");
190 }

316 void swap(MolecularBasisSetView& other) noexcept {
317 // As in PointSetView::swap, moving a view rebinds it

325 std::swap(m_atomic_numbers_, other.m_atomic_numbers_);
326 std::swap(m_purities_, other.m_purities_);
327 std::swap(m_orderings_, other.m_orderings_);
328 }
