169 template<typename BeginItr, typename EndItr>
170 static Grid from_range_(BeginItr begin, EndItr end) {
171 // push_back's TupleType (the types tried when the buffer isn't
172 // holding a concrete element type yet) defaults to
173 // wtf::default_fp_types (float/double/long double only), which
174 // would silently exclude every WTF-registered type this class's own
175 // docstring promises to support (e.g. Sigma's UQ types). Widen it
176 // explicitly to tensorwrapper's full floating_point_types list,
177 // which is exactly {float, double} plus, when compiled with
178 // ENABLE_SIGMA, every Sigma UQ type.
179 using tuple_type = chemist::types::floating_point_types;
180 wtf::buffer::FloatBuffer weights;
181 point_set_type points;
182 for(; begin != end; ++begin) {
183 weights.template push_back<tuple_type>(begin->get_weight());
184 points.push_back(*begin);
185 }
186 return Grid(std::move(weights), std::move(points));
187 }
