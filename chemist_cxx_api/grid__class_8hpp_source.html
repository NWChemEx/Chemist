263 template<typename BeginItr, typename EndItr>
264 static Grid from_range_(BeginItr begin, EndItr end) {
265 // push_back's TupleType (the types tried when the buffer isn't
266 // holding a concrete element type yet) defaults to
267 // wtf::default_fp_types (float/double/long double only), which
268 // would silently exclude every WTF-registered type this class's own
269 // docstring promises to support (e.g. Sigma's UQ types). Widen it
270 // explicitly to tensorwrapper's full floating_point_types list,
271 // which is exactly {float, double} plus, when compiled with
272 // ENABLE_SIGMA, every Sigma UQ type.
273 using tuple_type = chemist::types::floating_point_types;
274 wtf::buffer::FloatBuffer weights;
275 point_set_type points;
276 for(; begin != end; ++begin) {
277 weights.template push_back<tuple_type>(begin->get_weight());
278 points.push_back(*begin);
279 }
280 return Grid(std::move(weights), std::move(points));
281 }
