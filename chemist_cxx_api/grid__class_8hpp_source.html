135 template<typename BeginItr, typename EndItr>
136 static Grid from_range_(BeginItr begin, EndItr end) {
137 // push_back's TupleType (the types tried when the buffer isn't
138 // holding a concrete element type yet) defaults to
139 // wtf::default_fp_types (float/double/long double only), which
140 // would silently exclude every WTF-registered type this class's own
141 // docstring promises to support (e.g. Sigma's UQ types). Widen it
142 // explicitly to tensorwrapper's full floating_point_types list,
143 // which is exactly {float, double} plus, when compiled with
144 // ENABLE_SIGMA, every Sigma UQ type.
145 using tuple_type = chemist::types::floating_point_types;
146 wtf::buffer::FloatBuffer weights;
147 wtf::buffer::FloatBuffer coords;
148 for(; begin != end; ++begin) {
149 weights.template push_back<tuple_type>(begin->get_weight());
150 coords.template push_back<tuple_type>(begin->get_x());
151 coords.template push_back<tuple_type>(begin->get_y());
152 coords.template push_back<tuple_type>(begin->get_z());
153 }
154 return Grid(std::move(weights), std::move(coords));
155 }
