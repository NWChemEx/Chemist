225 template<typename Archive>
226 void load(Archive& ar) {
227 int purity{}, ordering{};
228 ar(purity, ordering);
229 name_type name;
230 atomic_number_type z{};
231 ar(name, z);
232 center_type center;
233 center.load(ar);
234 AtomicBasisSet buffer(std::move(name), z, std::move(center),
235 static_cast<ShellPurity>(purity),
236 static_cast<AOOrdering>(ordering));
237 size_type n_shells{};
238 ar(n_shells);
239 for(size_type i = 0; i < n_shells; ++i) {
240 angular_momentum_type l{};
241 size_type n_prims{};
242 ar(l, n_prims);
243 buffer_type cs, es;
244 for(size_type j = 0; j < n_prims; ++j) {
245 cs.template push_back<fp_types>(detail_::load_float(ar));
246 es.template push_back<fp_types>(detail_::load_float(ar));
247 }
248 buffer.m_pimpl_->add_shell(l, cs.as_view(), es.as_view());
249 }
250 swap(buffer);
251 }
