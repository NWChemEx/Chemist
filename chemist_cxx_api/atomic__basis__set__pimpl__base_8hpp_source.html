153 size_type n_aos() const noexcept {
154 const bool pure = purity() == ShellPurity::pure;
155 size_type n = 0;
156 for(size_type i = 0; i < size(); ++i) {
157 const auto l = l_(i);
158 n += pure ? 2 * l + 1 : (l + 1) * (l + 2) / 2;
159 }
160 return n;

177 void add_shell(angular_momentum_type l, const_buffer_reference coefficients,
178 const_buffer_reference exponents) {
179 if(coefficients.size() != exponents.size())
180 throw std::invalid_argument(
181 "chemist::experimental::AtomicBasisSet: the coefficient and "
182 "exponent arrays must be the same length.");
183 add_shell_(l, std::move(coefficients), std::move(exponents));
184 }

199 void push_back(const_shell_view_reference shell) {
200 const bool pure = purity() == ShellPurity::pure;
201 if(shell.is_pure() != pure)
202 throw_bad_shell_("its purity differs from that of the set");
203 if(!is_same_ordering_(shell))
204 throw_bad_shell_("its AO ordering differs from that of the set");
205 if(shell.get_center() != std::as_const(*this).center())
206 throw_bad_shell_("its center differs from that of the set");
207 auto cg = shell.get_contracted_gaussian();
208 add_shell(shell.get_l(), cg.get_coefficient_buffer(),
209 cg.get_exponent_buffer());
210 }

335 bool are_equal(const AtomicBasisSetPIMPLBase& rhs) const {
336 if(purity() != rhs.purity() || ordering() != rhs.ordering())
337 return false;
338 if(name() != rhs.name() || atomic_number() != rhs.atomic_number())
339 return false;
340 if(center() != rhs.center() || size() != rhs.size()) return false;
341 for(size_type i = 0; i < size(); ++i)
342 if(!at_(i)->are_equal(*rhs.at_(i))) return false;
343 return true;
344 }
