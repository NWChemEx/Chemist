150 size_type n_aos() const noexcept {
151 const bool pure = purity() == ShellPurity::pure;
152 size_type n = 0;
153 for(size_type i = 0; i < size(); ++i) {
154 const auto l = l_(i);
155 n += pure ? 2 * l + 1 : (l + 1) * (l + 2) / 2;

174 void add_shell(angular_momentum_type l, const_buffer_reference coefficients,

176 if(coefficients.size() != exponents.size())
177 throw std::invalid_argument(
178 "chemist::experimental::AtomicBasisSet: the coefficient and "
179 "exponent arrays must be the same length.");
180 add_shell_(l, std::move(coefficients), std::move(exponents));
181 }

196 void push_back(const AOShellView& shell) {
197 const bool pure = purity() == ShellPurity::pure;
198 if(shell.is_pure() != pure)
199 throw_bad_shell_("its purity differs from that of the set");
200 if(!is_same_ordering_(shell))
201 throw_bad_shell_("its AO ordering differs from that of the set");
202 if(shell.get_center() != std::as_const(*this).center())
203 throw_bad_shell_("its center differs from that of the set");
204 auto cg = shell.get_contracted_gaussian();
205 add_shell(shell.get_l(), cg.get_coefficient_buffer(),
206 cg.get_exponent_buffer());
207 }

332 bool are_equal(const AtomicBasisSetPIMPLBase& rhs) const {
333 if(purity() != rhs.purity() || ordering() != rhs.ordering())
334 return false;
335 if(name() != rhs.name() || atomic_number() != rhs.atomic_number())
336 return false;
337 if(center() != rhs.center() || size() != rhs.size()) return false;
338 for(size_type i = 0; i < size(); ++i)
339 if(!at_(i)->are_equal(*rhs.at_(i))) return false;
340 return true;
341 }

390 if(i < size()) return;
391 throw std::out_of_range(
392 "chemist::experimental::AtomicBasisSet: shell offset " +
393 std::to_string(i) + " is out of range for a set with " +
394 std::to_string(size()) + " shells.");
395 }
