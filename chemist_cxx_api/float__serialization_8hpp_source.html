63template<typename Archive>
64void save_float(Archive& ar, const_float_reference value) {
65 auto visitor = [&ar](const auto& v) {
66 using value_type = std::decay_t<decltype(v)>;
67 ar(std::string(wtf::type_traits::type_name_v<value_type>));
68 if constexpr(tensorwrapper::types::is_uq_type_v<value_type>) {
69 throw std::runtime_error(
70 "chemist::experimental: serializing uncertainty-quantification "
71 "floating-point types is not supported.");
72 } else {
73 ar(v);
74 }
75 };
76 wtf::fp::visit_float_view<serialization_fp_types>(visitor, value);
77}

87template<typename T, typename Archive>
88void load_one(Archive& ar, const std::string& name,
89 std::optional<float_type>& rv) {
90 if(rv.has_value()) return;
91 if(name != std::string(wtf::type_traits::type_name_v<T>)) return;
92 if constexpr(tensorwrapper::types::is_uq_type_v<T>) {
93 throw std::runtime_error(
94 "chemist::experimental: deserializing uncertainty-quantification "
95 "floating-point types is not supported.");
96 } else {
97 T value{};
98 ar(value);
99 rv.emplace(value);
100 }
101}

104template<typename Archive, typename... Ts>
105void load_each(Archive& ar, const std::string& name,
106 std::optional<float_type>& rv, const std::tuple<Ts...>*) {
107 (load_one<Ts>(ar, name, rv), ...);

125template<typename Archive>
126float_type load_float(Archive& ar) {
127 std::string name;
128 ar(name);
129
130 std::optional<float_type> rv;
131 const serialization_fp_types* tag = nullptr;
132 detail_impl_::load_each(ar, name, rv, tag);
133
134 if(!rv.has_value()) {
135 throw std::runtime_error(
136 "chemist::experimental: archive names an unknown floating-point "
137 "type: " +
138 name);
139 }
140 return std::move(*rv);
141}
