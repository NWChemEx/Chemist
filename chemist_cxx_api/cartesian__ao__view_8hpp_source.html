178 CartesianAOView(contracted_gaussian_reference cg, angular_index_type i,
179 angular_index_type j, angular_index_type k) :
180 m_cg_(std::move(cg)), m_i_(i), m_j_(j), m_k_(k) {

300 ao_type as_cartesian_ao() const {
301 const auto n = m_cg_.size();
302 std::vector<typename ao_type::coord_type> cs, es;
303 cs.reserve(n);
304 es.reserve(n);
305 for(size_type p = 0; p < n; ++p) {
306 cs.push_back(detail_::copy(m_cg_[p].get_coefficient()));
307 es.push_back(detail_::copy(m_cg_[p].get_exponent()));
308 }
309 return ao_type(cs.begin(), cs.end(), es.begin(), es.end(), m_i_, m_j_,
310 m_k_, m_cg_.get_center().as_point());
311 }
312
320 void swap(CartesianAOView& other) noexcept {
321 m_cg_.swap(other.m_cg_);
322 std::swap(m_i_, other.m_i_);
323 std::swap(m_j_, other.m_j_);
324 std::swap(m_k_, other.m_k_);
325 }

327private:
329 template<typename OtherDerived, typename OtherCAO>
330 void assign_(const CartesianAOCommon<OtherDerived, OtherCAO>& rhs) {
331 if(m_i_ != rhs.get_i() || m_j_ != rhs.get_j() || m_k_ != rhs.get_k())
332 throw std::runtime_error(
333 "chemist::experimental::CartesianAOView: can not assign a "
334 "Cartesian AO with different powers through a view. The powers "
335 "are not aliased, so writing them would leave the Cartesian AO "
336 "*this aliases disagreeing with *this.");
337 m_cg_ = rhs.get_contracted_gaussian();
338 }
