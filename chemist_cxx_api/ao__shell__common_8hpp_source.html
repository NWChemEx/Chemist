256template<typename TargetType, typename BaseType>
257const TargetType& checked_shell_downcast(const BaseType& shell) {
258 if(const auto* pshell = dynamic_cast<const TargetType*>(&shell))
259 return *pshell;
260 const bool want_pure =
261 ChemistClassTraits<AOShell<typename TargetType::ao_type>>::is_pure;
262 throw std::invalid_argument(
263 std::string("chemist::experimental: expected a ") +
264 (want_pure ? "pure" : "Cartesian") + " shell, but the shell is " +
265 (shell.is_pure() ? "pure" : "Cartesian") + ".");
266}
