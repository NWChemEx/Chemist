364 size_type n_aos() const noexcept {
365 const auto ls = downcast_().l_data_();
366 const auto purities = downcast_().purity_data_();
367 size_type n = 0;
368 for(size_type a = 0; a < size(); ++a) {
369 const bool pure = purities[a] == ShellPurity::pure;
370 const auto [b, e] = shell_range_(a);
371 for(auto s = b; s < e; ++s) {
372 const auto l = ls[s];
373 n += pure ? 2 * l + 1 : (l + 1) * (l + 2) / 2;
374 }
375 }
376 return n;
377 }

562 template<typename OtherDerived, typename OtherType>
563 void assign_(const MolecularBasisSetCommon<OtherDerived, OtherType>& rhs) {
564 static_assert(!is_const, "Can not assign through a read-only view.");
565 if(size() != rhs.size() || n_shells() != rhs.n_shells())
566 throw_shape_mismatch_();
567 const auto& d = downcast_();
568 const auto& rd = rhs.downcast_();
569 for(size_type a = 0; a < size(); ++a) {
570 if(d.purity_data_()[a] != rd.purity_data_()[a] ||
571 d.ordering_data_()[a] != rd.ordering_data_()[a] ||
572 shell_range_(a) != rhs.shell_range_(a))
573 throw_shape_mismatch_();
574 }
575 for(size_type s = 0; s < n_shells(); ++s)
576 if(primitive_range_(s) != rhs.primitive_range_(s))
577 throw_shape_mismatch_();
578
579 for(size_type a = 0; a < size(); ++a) {
580 auto atom = at_(a);
581 atom = rhs.at_(a);
582 }
583 }
584

587 [[noreturn]] static void throw_shape_mismatch_() {
588 throw std::runtime_error(
589 "chemist::experimental::MolecularBasisSetView: can not assign a set "
590 "with a different number of atoms, shell types, number of shells "
591 "per atom, or number of primitives per shell through a view.");
592 }

664 static size_type last_at_or_before_(SpanType offsets,
665 size_type i) noexcept {

677 template<typename ViewType, typename BufferType, typename LSpan,
678 typename CenterSet, typename NameSpan, typename ZSpan>
679 ViewType make_atom_(size_type a, BufferType cs, BufferType es, LSpan ls,
680 CenterSet centers, NameSpan names, ZSpan zs) const {
681 const auto& d = downcast_();
682 const auto offsets = d.primitive_offset_data_();
683 const auto [s_begin, s_end] = shell_range_(a);
684 const auto n_shells_on_atom = s_end - s_begin;
685 // offsets has one more element than there are shells, so this is
686 // valid even for a trailing atom with no shells
687 const auto p_begin = offsets[s_begin] - offsets.front();
688 const auto n_prims_on_atom = offsets[s_end] - offsets[s_begin];
689 return ViewType(detail_::make_atomic_basis_set_view_pimpl(
690 d.purity_data_()[a], d.ordering_data_()[a],
691 detail_::slice_buffer(std::move(cs), p_begin, n_prims_on_atom),
692 detail_::slice_buffer(std::move(es), p_begin, n_prims_on_atom),
693 ls.subspan(s_begin, n_shells_on_atom),
694 offsets.subspan(s_begin, n_shells_on_atom + 1), centers[a], names[a],
695 zs[a]));
696 }

707 d.center_data_(), d.name_data_(), d.atomic_number_data_());
708 }

716 const_center_set_reference(d.center_data_()),
717 typename traits_type::const_name_span(d.name_data_()),
718 typename traits_type::const_atomic_number_span(
719 d.atomic_number_data_()));
720 }
