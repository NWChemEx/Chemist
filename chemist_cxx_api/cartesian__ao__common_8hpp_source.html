439 template<typename OtherDerived, typename OtherPoint>
440 numerical_value apply_monomial_(
441 numerical_value value,
442 const PointCommon<OtherDerived, OtherPoint>& r) const {
443 const Point dr = r - get_center();
444 auto multiply_in = [&value](detail_::const_float_reference coord,
445 angular_index_type power) {
446 for(angular_index_type p = 0; p < power; ++p)
447 value = detail_::multiply(as_view_(value), coord);
448 };
449 multiply_in(dr.get_x(), get_i());
450 multiply_in(dr.get_y(), get_j());
451 multiply_in(dr.get_z(), get_k());
452 return value;
453 }
