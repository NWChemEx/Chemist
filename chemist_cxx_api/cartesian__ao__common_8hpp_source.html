485 template<typename OtherDerived, typename OtherPoint>
486 numerical_value apply_monomial_(
487 numerical_value value,
488 const PointCommon<OtherDerived, OtherPoint>& r) const {
489 const Point dr = r - get_center();
490 auto multiply_in = [&value](detail_::const_float_reference coord,
491 angular_index_type power) {
492 for(angular_index_type p = 0; p < power; ++p)
493 value = detail_::multiply(as_view_(value), coord);
494 };
495 multiply_in(dr.get_x(), get_i());
496 multiply_in(dr.get_y(), get_j());
497 multiply_in(dr.get_z(), get_k());
498 return value;
499 }
