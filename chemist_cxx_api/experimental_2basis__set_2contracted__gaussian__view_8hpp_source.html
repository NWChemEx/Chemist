259 contracted_gaussian_type as_contracted_gaussian() const {
260 const auto n = this->size();
261 std::vector<coefficient_type> cs;
262 std::vector<exponent_type> es;
263 cs.reserve(n);
264 es.reserve(n);
265 for(size_type i = 0; i < n; ++i) {
266 cs.push_back(detail_::copy(m_coefficients_.at(i)));
267 es.push_back(detail_::copy(m_exponents_.at(i)));
268 }
269 return contracted_gaussian_type(cs.begin(), cs.end(), es.begin(),
270 es.end(), *m_l_, m_center_.as_point());
271 }

280 void swap(ContractedGaussianView& other) noexcept {
281 std::swap(m_coefficients_, other.m_coefficients_);
282 std::swap(m_exponents_, other.m_exponents_);
283 std::swap(m_l_, other.m_l_);
284 m_center_.swap(other.m_center_);
285 }

287private:
289 template<typename OtherDerived, typename OtherCG>
290 void assign_(const ContractedGaussianCommon<OtherDerived, OtherCG>& rhs) {
291 const auto n = this->size();
292 if(n != rhs.size())
293 throw std::runtime_error(
294 "chemist::experimental::ContractedGaussianView: can not assign "
295 "a contraction of a different size through a view.");
296 for(size_type i = 0; i < n; ++i) (*this)[i] = rhs[i];
297 *m_l_ = rhs.get_l();
298 m_center_ = rhs.get_center();
299 }
