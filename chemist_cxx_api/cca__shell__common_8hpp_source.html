236 template<typename U = ShellType>
237 enable_if_cartesian_t<U, cartesian_powers_type> cartesian_powers(
238 size_type i) const {
239 check_offset_(i);
240 const auto l = get_l();
241 size_type offset{0};
242 for(angular_index_type x = l + 1; x-- > 0;) {
243 // With x fixed there are l - x + 1 AOs, one per y in [l-x, 0].
244 const auto n_in_block = l - x + 1;
245 if(i < offset + n_in_block) {
246 const angular_index_type y = (l - x) - (i - offset);
247 return cartesian_powers_type{x, y, l - x - y};
248 }
249 offset += n_in_block;
250 }
251 // Unreachable, since check_offset_ bounds i by the size of the shell
252 throw std::logic_error(
253 "chemist::experimental::CCAShell: ran off the end of the shell.");
254 }
