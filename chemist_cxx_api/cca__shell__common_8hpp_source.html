211 template<typename U = ShellType>
212 enable_if_cartesian_t<U, cartesian_powers_type> cartesian_powers(
213 size_type i) const {
214 check_offset_(i);
215 const auto l = get_l();
216 size_type offset{0};
217 for(angular_index_type x = l + 1; x-- > 0;) {
218 // With x fixed there are l - x + 1 AOs, one per y in [l-x, 0].
219 const auto n_in_block = l - x + 1;
220 if(i < offset + n_in_block) {
221 const angular_index_type y = (l - x) - (i - offset);
222 return cartesian_powers_type{x, y, l - x - y};
223 }
224 offset += n_in_block;
225 }
226 // Unreachable, since check_offset_ bounds i by the size of the shell
227 throw std::logic_error(
228 "chemist::experimental::CCAShell: ran off the end of the shell.");
229 }
