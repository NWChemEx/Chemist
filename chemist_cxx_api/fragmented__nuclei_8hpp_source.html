478template<typename NucleiType>
479template<typename BeginItr, typename EndItr>
480void FragmentedNuclei<NucleiType>::insert(BeginItr begin, EndItr end) {
481 // If we put the arguments directly into a vector this is what would result
482 using vec_t = decltype(std::vector(begin, end));
483
484 // These would be the elements
485 using input_type = typename vec_t::value_type;
486
487 // Were we given indices or nuclei?
488 if constexpr(std::is_integral_v<input_type>) {
489 nucleus_index_set inputs(begin, end);
490 insert(std::move(inputs));
491 } else {
492 // Assumed to be Nucleus or NucleusView objects
493 using input_t = typename const_reference::reference_container;
494 input_t inputs(begin, end);
495 insert(const_reference(inputs));
496 }
497}
