397 template<typename Archive>
398 void save(Archive& ar) const {
399 const auto& p = pimpl_();
400 ar(static_cast<int>(p.purity()), static_cast<int>(p.ordering()));
401 ar(p.name(), p.atomic_number());
402 p.center().save(ar);
403 const auto n_shells = p.size();
404 ar(n_shells);
405 const auto cs = p.coefficient_buffer();
406 const auto es = p.exponent_buffer();
407 for(size_type i = 0; i < n_shells; ++i) {
408 const auto [b, e] = p.primitive_range(i);
409 ar(p.get_l(i), e - b);
410 for(auto j = b; j < e; ++j) {
411 detail_::save_float(ar, cs.at(j));
412 detail_::save_float(ar, es.at(j));
413 }
414 }
415 }

442 template<typename OtherDerived, typename OtherType>
443 void assign_(const AtomicBasisSetCommon<OtherDerived, OtherType>& rhs) {
444 static_assert(!is_const, "Can not assign through a read-only view.");
445 auto& p = pimpl_();
446 const auto& rp = rhs.pimpl_();
447 if(&p == &rp) return;
448 if(p.purity() != rp.purity() || p.ordering() != rp.ordering() ||
449 p.size() != rp.size())
450 throw_shape_mismatch_();
451 for(size_type i = 0; i < p.size(); ++i)
452 if(p.primitive_range(i) != rp.primitive_range(i))
453 throw_shape_mismatch_();
454
455 p.name() = rp.name();
456 p.atomic_number() = rp.atomic_number();
457 p.center() = rp.center();
458 for(size_type i = 0; i < p.size(); ++i) p.set_l(i, rp.get_l(i));
459 auto cs = p.coefficient_buffer();
460 auto es = p.exponent_buffer();
461 const auto rcs = rp.coefficient_buffer();
462 const auto res = rp.exponent_buffer();
463 for(size_type i = 0; i < p.n_primitives(); ++i) {
464 detail_::assign(cs.at(i), rcs.at(i));
465 detail_::assign(es.at(i), res.at(i));
466 }
467 }
468

473 "chemist::experimental::AtomicBasisSetView: can not assign a set "
474 "with a different shell type, number of shells, or number of "
475 "primitives per shell through a view.");
476 }
