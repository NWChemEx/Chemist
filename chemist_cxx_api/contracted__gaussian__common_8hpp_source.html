251 template<typename OtherDerived, typename OtherPoint>
252 numerical_value evaluate(
253 const PointCommon<OtherDerived, OtherPoint>& r) const {
254 const auto n = this->size();
255 if(n == 0) return numerical_value(0.0);
256 auto acc = (*this)[0].evaluate(r);
257 for(size_type i = 1; i < n; ++i) {
258 auto term = (*this)[i].evaluate(r);
259 acc = detail_::add(as_view_(acc), as_view_(term));
260 }
261 return acc;
262 }

271 template<typename OtherDerived, typename OtherPointSet>
272 numerical_vector evaluate(
273 const PointSetCommon<OtherDerived, OtherPointSet>& points) const {
274 numerical_vector rv;
275 const auto n = points.size();
276 rv.reserve(n);
277 for(size_type i = 0; i < n; ++i) rv.push_back(evaluate(points[i]));
278 return rv;
279 }

315 template<typename OtherDerived, typename OtherPoint>
316 numerical_value normalized_evaluate(
317 const PointCommon<OtherDerived, OtherPoint>& r) const {
318 const auto n = this->size();
319 if(n == 0) return numerical_value(0.0);
320 auto n_g = normalization_constant();
321 auto acc = (*this)[0].normalized_evaluate(r);
322 for(size_type i = 1; i < n; ++i) {
323 auto term = (*this)[i].normalized_evaluate(r);
324 acc = detail_::add(as_view_(acc), as_view_(term));
325 }
326 return detail_::multiply(as_view_(n_g), as_view_(acc));
327 }
