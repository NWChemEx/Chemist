159 template<typename CoefBeginItr, typename CoefEndItr, typename ExpBeginItr,
160 typename ExpEndItr>
161 ContractedGaussian(CoefBeginItr&& cbegin, CoefEndItr&& cend,
162 ExpBeginItr&& ebegin, ExpEndItr&& eend,
163 angular_momentum_type l, center_type center) :
164 m_l_(l), m_center_(std::move(center)) {
165 for(auto itr = cbegin; itr != cend; ++itr)
166 m_coefficients_.template push_back<fp_types>(*itr);
167 for(auto itr = ebegin; itr != eend; ++itr)
168 m_exponents_.template push_back<fp_types>(*itr);
169 if(m_coefficients_.size() == m_exponents_.size()) return;
170 throw std::invalid_argument(
171 "chemist::experimental::ContractedGaussian: the coefficient and "
172 "exponent ranges must be the same length.");
173 }

220 template<typename Archive>
221 void load(Archive& ar) {
222 size_type n{};
223 ar(n);
224 angular_momentum_type l{};
225 ar(l);
226 center_type center;
227 center.load(ar);
228
229 buffer_type coefficients, exponents;
230 for(size_type i = 0; i < n; ++i) {
231 coefficients.template push_back<fp_types>(detail_::load_float(ar));
232 exponents.template push_back<fp_types>(detail_::load_float(ar));
233 }
234
235 ContractedGaussian buffer;
236 buffer.m_l_ = l;
237 buffer.m_center_ = std::move(center);
238 buffer.m_coefficients_ = std::move(coefficients);
239 buffer.m_exponents_ = std::move(exponents);
240 swap(buffer);
241 }
